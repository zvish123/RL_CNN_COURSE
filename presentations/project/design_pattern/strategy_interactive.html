class FileSender:
    def send(self, data: bytes, method: str):

            processed = data
        elif method == "compress":

06 / בניית הממשק

ממשק משותף עם ABC

from abc import ABC, abstractmethod

class ProcessingStrategy(ABC):
    @abstractmethod
    def process(self, data: bytes) -> bytes:
        pass

החוזה מחייב כל מחלקת Strategy לממש את process.

מה יקרה אם מחלקה שיורשת מ־ProcessingStrategy לא תממש את process?

לא ניתן ליצור ממנה מופע רגיל כל עוד היא נותרת מחלקה מופשטת. Python תזרוק TypeError בעת ניסיון ליצור מופע.
אפשר ליישם Strategy גם באמצעות duck typing או Protocol. כאן משתמשים ב־ABC כדי להמחיש חוזה ברור.
07 / אסטרטגיות קונקרטיות

שתי דרכים לאותה פעולה

import zlib

class NoProcessing(ProcessingStrategy):
    def process(self, data: bytes) -> bytes:
        return data

class CompressionStrategy(ProcessingStrategy):
    def process(self, data: bytes) -> bytes:
        return zlib.compress(data)

האם FileSender צריך לייבא את zlib?

לא. הידע על הדחיסה נמצא ב־CompressionStrategy. FileSender מכיר רק את הממשק המשותף.
08 / Context

המחלקה המרכזית נשארת פשוטה

class FileSender:
    def __init__(self, strategy: ProcessingStrategy):
        self.strategy = strategy

    def set_strategy(self, strategy: ProcessingStrategy):
        self.strategy = strategy

    def send(self, data: bytes) -> bytes:
        return self.strategy.process(data)
Composition over inheritance: ה־Context מחזיק אסטרטגיה כאובייקט, ולא יורש מכל אלגוריתם.

איזו שורה מאפשרת להחליף התנהגות בזמן ריצה?

self.strategy = strategy בתוך set_strategy — החלפת האובייקט משנה את הפעולה העתידית של send.

12 / איך מחליטים?

מתי כדאי לבחור Strategy?

  • קיימים כמה אלגוריתמים חלופיים לאותה פעולה.
  • צריך להחליף ביניהם בזמן ריצה או לפי הגדרה.
  • צפויות אסטרטגיות נוספות בעתיד.
  • רוצים לבדוק כל אלגוריתם בנפרד.
  • רוצים לצמצם תלות של הקוד המרכזי במימושים.

באיזה תרחיש Strategy מתאימה במיוחד?

זיהוי אנומליות יכול להשתמש בסף קבוע, ממוצע או כללים — כולם מימושים חלופיים של analyze.
13 / הימנעות ממורכבות מיותרת

מתי Strategy אינה הבחירה הטובה?

שתי אפשרויות פשוטות וקבועות

אם אין צפי להרחבה והלוגיקה קצרה, if עשוי להיות קריא יותר.

שלבים משלימים

אם צריך גם דחיסה וגם הצפנה ברצף, ייתכן ש־Pipeline / Decorator מתאים יותר מהצגתן כחלופות.

מערכת קטנה מחשבת מחיר עם הנחה קבועה אחת. מה עדיף?

תבניות הן כלי לפתרון בעיות תכנון, לא יעד בפני עצמו. אל תוסיפו מחלקות ללא צורך.

15 / אתגר יישום

NetworkMonitor — אתם המתכננים

מערכת ניטור תעבורת רשת צריכה לבחור בין אלגוריתמים שונים לזיהוי חריגה.

ThresholdStrategy

חריגה אם מספר הבקשות גדול מ־100.

AverageStrategy

חריגה אם המספר גדול מפי 2 מהממוצע.

RuleBasedStrategy

חריגה אם הבקשות מעל 80 וגם מספר ה־IP השונים מעל 20.

דרישות

  1. הגדירו ממשק DetectionStrategy עם detect(sample).
  2. ממשו לפחות שתי אסטרטגיות.
  3. ממשו NetworkMonitor שמאפשר להחליף אסטרטגיה.
  4. בדקו את אותה דגימה עם שתי אסטרטגיות שונות.
sample = {"requests": 120,
          "average": 70,
          "unique_ips": 25}
התחילו מ־class DetectionStrategy(ABC). כל אסטרטגיה תחזיר bool; ה־Context יקרא רק ל־detect.
from abc import ABC, abstractmethod

class DetectionStrategy(ABC):
    @abstractmethod
    def detect(self, sample: dict) -> bool:
        pass

class ThresholdStrategy(DetectionStrategy):
    def detect(self, sample):
        return sample["requests"] > 100

class AverageStrategy(DetectionStrategy):
    def detect(self, sample):
        return sample["requests"] > 2 * sample["average"]

class RuleBasedStrategy(DetectionStrategy):
    def detect(self, sample):
        return (sample["requests"] > 80
                and sample["unique_ips"] > 20)

class NetworkMonitor:
    def __init__(self, strategy: DetectionStrategy):
        self.strategy = strategy

    def set_strategy(self, strategy: DetectionStrategy):
        self.strategy = strategy

    def analyze(self, sample: dict) -> bool:
        return self.strategy.detect(sample)

sample = {"requests": 120, "average": 70,
          "unique_ips": 25}
monitor = NetworkMonitor(ThresholdStrategy())
print(monitor.analyze(sample))  # True
monitor.set_strategy(AverageStrategy())
print(monitor.analyze(sample))  # False
monitor.set_strategy(RuleBasedStrategy())
print(monitor.analyze(sample))  # True

17 / כרטיס יציאה

מה תיקחו לפרויקט הגמר?

שלוש שאלות לפני בחירה ב־Strategy

  1. האם יש כמה אלגוריתמים חלופיים לאותה פעולה?
  2. האם צריך להחליף ביניהם או להוסיף חדשים?
  3. האם הפרדה למחלקות תפשט את הקוד יותר משהיא תסבך אותו?

כתבו דוגמה אחת מפרויקט הסיום שבה Strategy עשויה לעזור, והסבירו מדוע.

במערכת ניטור רשת קיימות שיטות שונות לזיהוי תעבורה חשודה: סף קבוע, השוואה לממוצע וכללים. Strategy תאפשר להחליף את שיטת הזיהוי בלי לשנות את מחלקת NetworkMonitor.
