מהי תבנית Proxy?

08 · Protection Proxy

בדיקת הרשאות לפני גישה

class FileAccessProxy(FileStorage):
    def __init__(self, storage, user, permissions):
        self.storage = storage
        self.user = user
        self.permissions = permissions

    def download(self, filename: str) -> bytes:
        if not self.permissions.can_read(
            self.user, filename
        ):
            raise PermissionError("Access denied")
        return self.storage.download(filename)
המשתמש חייב להיקבע על סמך אימות מהימן בצד השרת — לא מתוך שדה username שהלקוח יכול לזייף.

מה בודקים?

כמה פעמים נוצר האובייקט לאחר שלוש קריאות?

פעם אחת בלבד; הקריאות הבאות משתמשות באותו אובייקט.
class LazyStorageProxy(FileStorage):
    def __init__(self, factory):
        self.factory = factory
        self.real = None

    def download(self, filename):
        if self.real is None:
            self.real = self.factory()
        return self.real.download(filename)

שמרו לכל IP זוג (expiry_time, result). בדקו תוקף לפני שימוש בתוצאה.
השתמשו ב־time.monotonic() למדידת חלוף זמן, והגדירו האם בקשת Cache Hit צורכת מכסה.
import time

class ThreatIntelligenceProxy:
    def __init__(self, api, ttl=60, min_interval=1):
        self.api = api
        self.ttl = ttl
        self.min_interval = min_interval
        self.cache = {}
        self.last_call = float('-inf')

    def check_ip(self, ip):
        if not isinstance(ip, str) or not ip.strip():
            raise ValueError("Invalid IP")
        now = time.monotonic()
        entry = self.cache.get(ip)
        if entry and now < entry[0]:
            return entry[1].copy()
        if now - self.last_call < self.min_interval:
            raise RuntimeError("Rate limit exceeded")
        self.last_call = now
        result = self.api.check_ip(ip)
        self.cache[ip] = (time.monotonic()+self.ttl,
                          result.copy())
        return result.copy()

פתרון לימודי: מגבלת קצב פשוטה לתהליך יחיד; אין כאן טיפול בריבוי תהליכים, retries או ולידציה מלאה של כתובת IP.
