02 · המימוש הראשון: תנאים

class SecureVault:
    def access_files(self):
        if self.state == "LOCKED":
            return "Access denied"
        elif self.state == "BLOCKED":
            return "Vault blocked"
        elif self.state == "UNLOCKED":
            return "Files available"
מה הבעיה כאן? אל תחשפו את התשובה לפני דיון.

07 · חוזה משותף עם ABC

from abc import ABC, abstractmethod

class VaultState(ABC):
    @abstractmethod
    def enter_password(self, vault, password):
        pass

    @abstractmethod
    def access_files(self, vault):
        pass

    @abstractmethod
    def lock(self, vault):
        pass

    @abstractmethod
    def admin_reset(self, vault):
        pass

המתודה מקבלת vault כדי לקרוא נתונים משותפים ולבצע מעבר מצב. בפייתון אפשר גם להשתמש ב־Protocol או במימושים פשוטים ללא ABC.

08 · מימוש LockedState

class LockedState(VaultState):
    def enter_password(self, vault, password):
        if vault.check_password(password):
            vault.attempts = 0
            vault.set_state(UnlockedState())
        else:
            vault.attempts += 1
            if vault.attempts >= 3:
                vault.set_state(BlockedState())

    def access_files(self, vault):
        return "Access denied"

    def lock(self, vault):
        return "Already locked"

    def admin_reset(self, vault):
        return "No reset needed"
מה משתנה לאחר ניסיון שגוי שני? ומה משתנה לאחר השלישי?
אחרי השני: attempts == 2 והמצב Locked. אחרי השלישי: attempts == 3 והמצב Blocked.

09 · מצבים נוספים

class UnlockedState(VaultState):
    def enter_password(self, vault, password):
        return "Already open"

    def access_files(self, vault):
        return "Files available"

    def lock(self, vault):
        vault.set_state(LockedState())
        return "Locked"

    def admin_reset(self, vault):
        return "Not blocked"

class BlockedState(VaultState):
    def enter_password(self, vault, password):
        return "Blocked"

    def access_files(self, vault):
        return "Blocked"

    def lock(self, vault):
        return "Blocked"

    def admin_reset(self, vault):
        vault.attempts = 0
        vault.set_state(LockedState())
        return "Reset completed"

10 · מחלקת Context

class SecureVault:
    def __init__(self, password):
        self._password = password  # demo only
        self.attempts = 0
        self.state = LockedState()

    def check_password(self, password):
        return password == self._password

    def set_state(self, new_state):
        self.state = new_state

    def enter_password(self, password):
        return self.state.enter_password(self, password)

    def access_files(self):
        return self.state.access_files(self)

    def lock(self):
        return self.state.lock(self)

    def admin_reset(self):
        return self.state.admin_reset(self)

דוגמה לימודית בלבד: במערכת אמיתית אין לשמור סיסמה גלויה או לבצע אימות רגיש בצד לקוח.

13 · מתי כדאי לבחור State?

✓ התנהגות משתנה לפי מצב פנימי
✓ יש מעברים ואירועים מוגדרים
✓ תנאים דומים מופיעים במספר מתודות
✓ צפויים מצבים חדשים או כללי מעבר חדשים
תרחיש: כפתור אחד שמציג "פעיל" או "כבוי" בלבד, ללא פעולות נוספות. האם צריך State?
כנראה שלא. משתנה בוליאני ותנאי קצר יהיו פשוטים יותר. אין חובה להשתמש בתבנית בכל מצב.

15 · אתגר יישום: חיבור TCP מדומה

תכננו TCPConnection עם המצבים ClosedState, ConnectingState, ConnectedState.

Closed
connect() →
Connecting
success() →
Connected
  1. send(data) מותר רק במצב Connected.
  2. connect() מעביר מ־Closed ל־Connecting.
  3. connection_success() מעביר ל־Connected.
  4. connection_failed() מחזיר ל־Closed.
  5. close() מחזיר ל־Closed.

מודל מופשט לתרגול State, לא מכונת המצבים המלאה של TCP.

רמז ראשון: מבנה המחלקות

הגדירו ConnectionState עם פעולות משותפות. ה־Context ישמור state ויפנה אליו קריאות.

16 · אתגר קוד: בדקו את עצמכם

השלימו את המימוש לפני פתיחת הפתרון. חשבו אילו פעולות אינן חוקיות בכל מצב.

class TCPConnection:
    def __init__(self):
        self.state = ClosedState()

    def set_state(self, state):
        self.state = state

    def connect(self):
        return self.state.connect(self)

    def send(self, data):
        return self.state.send(self, data)
פתרון לדוגמה — חשפו רק לאחר ניסיון
class ConnectionState:
    def connect(self, c): return "Cannot connect"
    def success(self, c): return "Unexpected success"
    def failed(self, c): return "Unexpected failure"
    def send(self, c, data): return "Not connected"
    def close(self, c):
        c.set_state(ClosedState())
        return "Closed"

class ClosedState(ConnectionState):
    def connect(self, c):
        c.set_state(ConnectingState())
        return "Connecting"

class ConnectingState(ConnectionState):
    def success(self, c):
        c.set_state(ConnectedState())
        return "Connected"
    def failed(self, c):
        c.set_state(ClosedState())
        return "Connection failed"

class ConnectedState(ConnectionState):
    def send(self, c, data):
        return f"Sent {len(data)} bytes"

class TCPConnection:
    def __init__(self): self.state = ClosedState()
    def set_state(self, state): self.state = state
    def connect(self): return self.state.connect(self)
    def success(self): return self.state.success(self)
    def failed(self): return self.state.failed(self)
    def send(self, data): return self.state.send(self, data)
    def close(self): return self.state.close(self)

c = TCPConnection()
assert c.send(b"Hi") == "Not connected"
assert c.connect() == "Connecting"
assert c.success() == "Connected"
assert c.send(b"Hi") == "Sent 2 bytes"
assert c.close() == "Closed"
